CREATE SCHEMA IF NOT EXISTS public;

CREATE TABLE users (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    dog_name VARCHAR(80) NOT NULL,
    breed VARCHAR(120) DEFAULT NULL,
    chip_number VARCHAR(80) DEFAULT NULL,
    weight_lbs DECIMAL(5,2) DEFAULT NULL,
    recovery_key VARCHAR(10) NOT NULL,
    google_2fa_secret VARCHAR(32) DEFAULT NULL,
    is_2fa_enabled SMALLINT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE dogs (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    owner_user_id INTEGER NOT NULL,
    name VARCHAR(80) NOT NULL,
    breed VARCHAR(120) DEFAULT NULL,
    chip_number VARCHAR(80) DEFAULT NULL,
    weight_lbs DECIMAL(5,2) DEFAULT NULL,
    date_of_birth DATE DEFAULT NULL,
    birth_is_approximate SMALLINT DEFAULT 0,
    approx_age_years DECIMAL(4,1) DEFAULT NULL,
    notes TEXT DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ,
    CONSTRAINT fk_dogs_owner FOREIGN KEY (owner_user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE dog_handlers (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dog_id INTEGER NOT NULL,
    user_id INTEGER NOT NULL,
    invited_by_user_id INTEGER NOT NULL,
    role TEXT CHECK (role IN ('owner', 'collaborator')) DEFAULT 'collaborator',
    permission_level TEXT CHECK (permission_level IN ('view', 'edit')) DEFAULT 'edit',
    status TEXT CHECK (status IN ('accepted', 'revoked')) DEFAULT 'accepted',
    accepted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (dog_id, user_id),
    CONSTRAINT fk_dog_handlers_dog FOREIGN KEY (dog_id) REFERENCES dogs(id) ON DELETE CASCADE,
    CONSTRAINT fk_dog_handlers_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    CONSTRAINT fk_dog_handlers_inviter FOREIGN KEY (invited_by_user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE handler_handshakes (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dog_id INTEGER NOT NULL,
    code VARCHAR(12) NOT NULL UNIQUE,
    created_by_user_id INTEGER NOT NULL,
    requested_by_user_id INTEGER DEFAULT NULL,
    requested_permission TEXT CHECK (requested_permission IN ('view', 'edit')) DEFAULT 'edit',
    status TEXT CHECK (status IN ('open', 'requested', 'approved', 'declined', 'expired', 'revoked')) DEFAULT 'open',
    expires_at TIMESTAMP NOT NULL,
    requested_at TIMESTAMP DEFAULT NULL,
    decided_at TIMESTAMP DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_handshake_dog FOREIGN KEY (dog_id) REFERENCES dogs(id) ON DELETE CASCADE,
    CONSTRAINT fk_handshake_creator FOREIGN KEY (created_by_user_id) REFERENCES users(id) ON DELETE CASCADE,
    CONSTRAINT fk_handshake_requester FOREIGN KEY (requested_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);

CREATE TABLE dog_vets (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dog_id INTEGER NOT NULL,
    clinic_name VARCHAR(120) NOT NULL,
    vet_name VARCHAR(120) DEFAULT NULL,
    phone VARCHAR(40) DEFAULT NULL,
    address TEXT DEFAULT NULL,
    is_primary SMALLINT DEFAULT 0,
    notes TEXT DEFAULT NULL,
    created_by_user_id INTEGER NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_dog_vets_dog FOREIGN KEY (dog_id) REFERENCES dogs(id) ON DELETE CASCADE,
    CONSTRAINT fk_dog_vets_creator FOREIGN KEY (created_by_user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE dog_documents (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dog_id INTEGER NOT NULL,
    uploaded_by_user_id INTEGER NOT NULL,
    doc_type TEXT CHECK (doc_type IN ('vet_record', 'esa_letter', 'service_dog_letter')) NOT NULL,
    title VARCHAR(150) NOT NULL,
    provider_name VARCHAR(150) DEFAULT NULL,
    notes TEXT DEFAULT NULL,
    file_path VARCHAR(255) NOT NULL,
    mime_type VARCHAR(100) DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_dog_documents_dog FOREIGN KEY (dog_id) REFERENCES dogs(id) ON DELETE CASCADE,
    CONSTRAINT fk_dog_documents_uploader FOREIGN KEY (uploaded_by_user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE dog_vet_appointments (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dog_id INTEGER NOT NULL,
    dog_vet_id INTEGER DEFAULT NULL,
    created_by_user_id INTEGER NOT NULL,
    title VARCHAR(150) NOT NULL,
    appointment_at TIMESTAMP NOT NULL,
    reminder_at TIMESTAMP DEFAULT NULL,
    location_text VARCHAR(255) DEFAULT NULL,
    notes TEXT DEFAULT NULL,
    status TEXT CHECK (status IN ('scheduled', 'completed', 'cancelled')) DEFAULT 'scheduled',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_dog_appointments_dog FOREIGN KEY (dog_id) REFERENCES dogs(id) ON DELETE CASCADE,
    CONSTRAINT fk_dog_appointments_vet FOREIGN KEY (dog_vet_id) REFERENCES dog_vets(id) ON DELETE SET NULL,
    CONSTRAINT fk_dog_appointments_creator FOREIGN KEY (created_by_user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE daily_logs (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id INTEGER NOT NULL,
    dog_id INTEGER NOT NULL,
    log_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    location_name VARCHAR(100) NOT NULL,
    location_city_state VARCHAR(100) DEFAULT NULL,
    location_type TEXT CHECK (location_type IN ('In-Cab', 'Truck Stop', 'Shipper/Receiver', 'Public Store', 'Rest Area', 'Other')) DEFAULT 'Other',
    focus_level SMALLINT DEFAULT 3,
    skills_practiced JSONB,
    handler_notes TEXT,
    media_url VARCHAR(255) DEFAULT NULL,
    media_type TEXT CHECK (media_type IN ('image', 'video')) DEFAULT NULL,
    media_mime VARCHAR(100) DEFAULT NULL,
    latitude DECIMAL(10,7) DEFAULT NULL,
    longitude DECIMAL(10,7) DEFAULT NULL,
    CONSTRAINT fk_user_logs FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    CONSTRAINT fk_dog_logs FOREIGN KEY (dog_id) REFERENCES dogs(id) ON DELETE CASCADE,
    CONSTRAINT chk_focus_level CHECK (focus_level BETWEEN 1 AND 5)
);


CREATE TABLE dog_medications (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dog_id INTEGER NOT NULL,
    created_by_user_id INTEGER NOT NULL,
    medication_name VARCHAR(150) NOT NULL,
    dosage VARCHAR(120) DEFAULT NULL,
    instructions TEXT DEFAULT NULL,
    start_date DATE DEFAULT NULL,
    end_date DATE DEFAULT NULL,
    schedule_text VARCHAR(150) DEFAULT NULL,
    reminder_time TIMESTAMP DEFAULT NULL,
    refill_date DATE DEFAULT NULL,
    prescribing_provider VARCHAR(150) DEFAULT NULL,
    pharmacy_name VARCHAR(150) DEFAULT NULL,
    pharmacy_phone VARCHAR(40) DEFAULT NULL,
    status TEXT CHECK (status IN ('active', 'paused', 'completed')) DEFAULT 'active',
    notes TEXT DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_dog_medications_dog FOREIGN KEY (dog_id) REFERENCES dogs(id) ON DELETE CASCADE,
    CONSTRAINT fk_dog_medications_creator FOREIGN KEY (created_by_user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE dog_certification_items (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dog_id INTEGER NOT NULL,
    category VARCHAR(100) NOT NULL,
    item_name VARCHAR(180) NOT NULL,
    description TEXT DEFAULT NULL,
    is_required SMALLINT DEFAULT 1,
    sort_order INTEGER DEFAULT 0,
    status TEXT CHECK (status IN ('not_started', 'in_training', 'proficient')) DEFAULT 'not_started',
    last_assessed_at TIMESTAMP DEFAULT NULL,
    notes TEXT DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_dog_cert_items_dog FOREIGN KEY (dog_id) REFERENCES dogs(id) ON DELETE CASCADE
);

CREATE TABLE dog_certification_assessments (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dog_id INTEGER NOT NULL,
    assessed_by_user_id INTEGER NOT NULL,
    assessment_date DATE NOT NULL,
    public_access_score SMALLINT DEFAULT NULL,
    task_reliability_score SMALLINT DEFAULT NULL,
    obedience_score SMALLINT DEFAULT NULL,
    environmental_score SMALLINT DEFAULT NULL,
    notes TEXT DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_dog_cert_assess_dog FOREIGN KEY (dog_id) REFERENCES dogs(id) ON DELETE CASCADE,
    CONSTRAINT fk_dog_cert_assess_user FOREIGN KEY (assessed_by_user_id) REFERENCES users(id) ON DELETE CASCADE
);


CREATE TABLE dog_training_profiles (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dog_id INTEGER NOT NULL UNIQUE,
    created_by_user_id INTEGER NOT NULL,
    training_mode TEXT CHECK (training_mode IN ('self_training', 'professional_trainer', 'hybrid')) DEFAULT 'self_training',
    trainer_name VARCHAR(120) DEFAULT NULL,
    business_name VARCHAR(150) DEFAULT NULL,
    credentials TEXT DEFAULT NULL,
    trainer_phone VARCHAR(40) DEFAULT NULL,
    trainer_email VARCHAR(120) DEFAULT NULL,
    trainer_website VARCHAR(150) DEFAULT NULL,
    handler_experience VARCHAR(120) DEFAULT NULL,
    handler_goals TEXT DEFAULT NULL,
    candidate_stage TEXT CHECK (candidate_stage IN ('prospect', 'foundation', 'advanced', 'working', 'career_change', 'retired')) DEFAULT 'prospect',
    candidate_notes TEXT DEFAULT NULL,
    training_focus TEXT DEFAULT NULL,
    notes TEXT DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ,
    CONSTRAINT fk_dog_training_profile_dog FOREIGN KEY (dog_id) REFERENCES dogs(id) ON DELETE CASCADE,
    CONSTRAINT fk_dog_training_profile_user FOREIGN KEY (created_by_user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE dog_training_items (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dog_id INTEGER NOT NULL,
    category VARCHAR(100) NOT NULL,
    track_code VARCHAR(40) DEFAULT NULL,
    level SMALLINT DEFAULT 1,
    item_name VARCHAR(180) NOT NULL,
    description TEXT DEFAULT NULL,
    sort_order INTEGER DEFAULT 0,
    status TEXT CHECK (status IN ('not_started', 'in_progress', 'proofing', 'mastered')) DEFAULT 'not_started',
    last_worked_at TIMESTAMP DEFAULT NULL,
    notes TEXT DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_dog_training_items_dog FOREIGN KEY (dog_id) REFERENCES dogs(id) ON DELETE CASCADE
);


CREATE INDEX idx_user_logs ON daily_logs(user_id);
CREATE INDEX idx_dog_logs ON daily_logs(dog_id);
CREATE INDEX idx_log_date ON daily_logs(log_date);
CREATE INDEX idx_log_coordinates ON daily_logs(latitude, longitude);
CREATE INDEX idx_dog_owner ON dogs(owner_user_id);
CREATE INDEX idx_handshake_status ON handler_handshakes(status, expires_at);
CREATE INDEX idx_dog_documents_type ON dog_documents(dog_id, doc_type);
CREATE INDEX idx_dog_vet_appointments_when ON dog_vet_appointments(dog_id, appointment_at, reminder_at);

CREATE INDEX idx_dog_medications_due ON dog_medications(dog_id, status, reminder_time, refill_date);
CREATE INDEX idx_dog_cert_items_status ON dog_certification_items(dog_id, category, status);

CREATE INDEX idx_dog_training_items_status ON dog_training_items(dog_id, level, status);
